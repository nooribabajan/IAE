/**
 * Universal Refactoring Workbench — conservative HTML analyzer.
 *
 * Browser-only module; no dependencies. Import with:
 *   import { analyzeHtml } from "./analyzer.js";
 *
 * This is an advisory static analyzer, not a browser-equivalent validator.
 * It never edits the input, executes its scripts, or declares files safe to delete.
 */

const MAX_FINDINGS = 250;
const MAX_CSS_CONFLICTS = 40;

/** @typedef {{severity: "error"|"warning"|"info", title: string, detail: string, code?: string}} Finding */

/**
 * Analyze a single HTML document.
 * @param {string} source Raw HTML source.
 * @returns {{lines:number, elements:number, bytes:number, findings:Finding[], scripts:{total:number,inline:number,external:number}, styles:{inline:number,external:number}, comments:number, parserNote:string}}
 */
function analyzeHtml(source) {
  if (typeof source !== "string") {
    throw new TypeError("analyzeHtml(source) expects an HTML string.");
  }
  if (typeof DOMParser === "undefined") {
    throw new Error("This analyzer requires a browser with DOMParser support.");
  }

  const findings = [];
  const seen = new Set();
  const add = (severity, message, code = "HTML_AUDIT") => {
    const key = `${severity}\u0000${code}\u0000${message}`;
    if (seen.has(key) || findings.length >= MAX_FINDINGS) return;
    seen.add(key);
    const firstSentence = message.split(/(?<=[.!?])\s+/)[0] || message;
    findings.push({ severity, title: firstSentence.length > 92 ? `${firstSentence.slice(0, 89)}…` : firstSentence, detail: message, code });
  };

  const doc = new DOMParser().parseFromString(source, "text/html");
  const all = [...doc.querySelectorAll("*")];
  const lines = source.length ? source.split(/\r\n|\r|\n/).length : 0;
  const bytes = new TextEncoder().encode(source).length;
  const scriptNodes = [...doc.querySelectorAll("script")];
  const scripts = { total: scriptNodes.length, inline: scriptNodes.filter(node => !node.getAttribute("src")).length, external: scriptNodes.filter(node => Boolean(node.getAttribute("src"))).length };
  const styleNodesForCount = [...doc.querySelectorAll("style")];
  const styles = { inline: styleNodesForCount.length, external: doc.querySelectorAll('link[rel~="stylesheet"][href]').length };
  const comments = (source.match(/<!--[\s\S]*?-->/g) || []).length;

  // Preserve useful checks from the previous stable baseline.
  const sourceWithoutClosedComments = source.replace(/<!--[\s\S]*?-->/g, "");
  if (sourceWithoutClosedComments.includes("<!--")) {
    add("warning", "A possible unclosed HTML comment marker remains in the source. Template syntax may affect this heuristic; review manually.", "POSSIBLE_UNCLOSED_COMMENT");
  }
  if (source.includes("{{") || source.includes("{%") || source.includes("<%")) {
    add("info", "Template markers were detected. Preserve delimiters and source order during future transformations.", "TEMPLATE_SYNTAX");
  }
  if (bytes > 2 * 1024 * 1024) {
    add("info", "The HTML source exceeds 2 MiB. Keep transformations conservative and test on a copy.", "LARGE_FILE");
  }

  // Document-level checks.
  if (!/<\s*!doctype\s+html\s*>/i.test(source)) {
    add("warning", "Missing <!doctype html>; browsers may use quirks mode.");
  }
  if (!doc.documentElement.hasAttribute("lang") || !doc.documentElement.lang.trim()) {
    add("warning", "The <html> element has no language declaration (lang).");
  }
  if (!/<meta\b[^>]*\bcharset\s*=/i.test(source) &&
      !/<meta\b[^>]*http-equiv\s*=\s*["']?content-type/i.test(source)) {
    add("info", "No character-encoding declaration was found in the source.");
  }
  if (!doc.querySelector("title") || !doc.querySelector("title").textContent.trim()) {
    add("warning", "The document has no non-empty <title>.");
  }
  if (!doc.querySelector('meta[name="viewport" i]')) {
    add("info", "No viewport meta tag was found; mobile layout may not scale as intended.");
  }

  // Duplicate IDs: actual parsed elements, not a text-only regex.
  const ids = new Map();
  for (const el of all) {
    const id = el.getAttribute("id");
    if (id === null || id === "") continue;
    ids.set(id, (ids.get(id) || 0) + 1);
  }
  for (const [id, count] of ids) {
    if (count > 1) add("error", `Duplicate id="${id}" appears on ${count} elements; fragment links and selectors may target the wrong element.`, "DUPLICATE_ID");
  }
  const uniqueIds = new Set(ids.keys());
  for (const link of doc.querySelectorAll('a[href^="#"]')) {
    const fragment = link.getAttribute("href").slice(1);
    if (!fragment) continue;
    let decoded = fragment;
    try { decoded = decodeURIComponent(fragment); } catch (error) { decoded = fragment; }
    if (!uniqueIds.has(decoded)) add("warning", `An in-page link points to #${fragment}, but no matching id was found in this HTML document.`, "BROKEN_FRAGMENT");
  }

  // Accessibility and semantics.
  for (const img of doc.querySelectorAll("img")) {
    if (!img.hasAttribute("alt")) {
      add("warning", "An <img> has no alt attribute; add descriptive alt text or alt=\"\" if decorative.");
    }
    if (!img.hasAttribute("src") && !img.hasAttribute("srcset")) {
      add("warning", "An <img> has neither src nor srcset.");
    }
  }
  for (const frame of doc.querySelectorAll("iframe, frame")) {
    if (!frame.getAttribute("title")?.trim()) {
      add("warning", `A <${frame.tagName.toLowerCase()}> has no title for assistive technology.`);
    }
  }
  for (const control of doc.querySelectorAll("input, select, textarea")) {
    if (control.type === "hidden" || control.disabled) continue;
    const id = control.id;
    const hasLabel = (id && [...doc.querySelectorAll("label[for]")].some(label => label.htmlFor === id)) ||
      Boolean(control.closest("label")) ||
      Boolean(control.getAttribute("aria-label")?.trim()) ||
      Boolean(control.getAttribute("aria-labelledby")?.trim());
    if (!hasLabel) add("warning", `A <${control.tagName.toLowerCase()}${control.type ? ` type="${control.type}"` : ""}> may lack an accessible label.`);
  }
  for (const button of doc.querySelectorAll("button, input[type=button], input[type=submit], input[type=reset]")) {
    const name = button.tagName.toLowerCase() === "input"
      ? button.value
      : button.textContent.replace(/\s+/g, " ").trim();
    if (!name && !button.getAttribute("aria-label")?.trim() && !button.getAttribute("aria-labelledby")?.trim()) {
      add("warning", `A <${button.tagName.toLowerCase()}> has no obvious accessible name.`);
    }
  }
  for (const link of doc.querySelectorAll("a")) {
    if (!link.hasAttribute("href")) {
      add("info", "An <a> element has no href; confirm it is intentionally being used as a non-navigation element.");
    }
    if (link.getAttribute("target")?.toLowerCase() === "_blank") {
      const rel = new Set((link.getAttribute("rel") || "").toLowerCase().split(/\s+/).filter(Boolean));
      if (!rel.has("noopener")) add("info", 'A link uses target="_blank" without rel="noopener"; review opener isolation and intended behavior.');
    }
  }

  // Suspicious constructs: findings are contextual, not automatic deletion instructions.
  let inlineHandlerCount = 0;
  const handlerNames = new Set();
  for (const el of all) {
    for (const attr of [...el.attributes]) {
      if (/^on[a-z]+$/i.test(attr.name)) {
        inlineHandlerCount++;
        handlerNames.add(attr.name.toLowerCase());
      }
    }
  }
  if (inlineHandlerCount) {
    add("info", `Found ${inlineHandlerCount} inline event-handler attribute(s) (${[...handlerNames].sort().join(", ")}). These are not automatically defects; inspect their behavior before changing them.`);
  }

  for (const el of all) {
    for (const attrName of ["href", "src", "action", "formaction", "xlink:href"]) {
      if (!el.hasAttribute(attrName)) continue;
      const value = el.getAttribute(attrName).trim();
      if (/^javascript\s*:/i.test(value)) {
        add("warning", `A javascript: URL was found in ${el.tagName.toLowerCase()}[${attrName}]; verify necessity and security.`, "JAVASCRIPT_URL");
      }
      if (!value && ["src", "action", "formaction"].includes(attrName)) {
        add("info", `An empty ${attrName} attribute was found on <${el.tagName.toLowerCase()}>; verify intended browser behavior.`);
      }
    }
  }

  // Local references cannot be proven missing when only one HTML file is supplied.
  const localRefs = new Set();
  const externalRefs = new Set();
  const addReference = raw => {
    const value = String(raw || "").trim();
    if (!value || value.startsWith("#") || /^data:/i.test(value)) return;
    if (/^(?:[a-z][a-z\d+.-]*:|\/\/)/i.test(value)) {
      if (/^(?:https?:)?\/\//i.test(value)) externalRefs.add(value);
      return;
    }
    localRefs.add(value.split(/[?#]/, 1)[0]);
  };
  for (const el of all) {
    for (const attr of ["src", "href", "poster", "data", "action", "formaction"]) {
      if (el.hasAttribute(attr)) addReference(el.getAttribute(attr));
    }
    if (el.hasAttribute("srcset")) {
      for (const part of el.getAttribute("srcset").split(",")) addReference(part.trim().split(/\s+/)[0]);
    }
  }
  for (const style of doc.querySelectorAll("style")) {
    for (const match of style.textContent.matchAll(/url\(\s*["']?([^"')]+)["']?\s*\)/gi)) addReference(match[1]);
  }
  if (localRefs.size) {
    add("info", `${localRefs.size} relative/local resource reference(s) were found. Because only this HTML file is analyzed, their existence and deployment paths cannot be verified: ${[...localRefs].slice(0, 8).join(", ")}${localRefs.size > 8 ? ", …" : ""}.`);
  }
  if (externalRefs.size) {
    add("info", `${externalRefs.size} external HTTP(S) resource reference(s) were found. Network availability and remote content were not checked.`);
  }

  // CSS matching: analyze only CSS embedded in this HTML. External stylesheets are not fetched.
  // Report competing declarations, but never claim a definitive cascade winner.
  const cssResult = inspectEmbeddedCss(doc);
  for (const message of cssResult.messages.slice(0, MAX_CSS_CONFLICTS)) add("info", message);
  if (cssResult.externalStylesheets > 0) {
    add("info", `${cssResult.externalStylesheets} linked stylesheet(s) were found but not loaded. CSS matches from those files cannot be assessed from this single-file input.`);
  }
  if (cssResult.unsupportedRules > 0) {
    add("info", `${cssResult.unsupportedRules} CSS rule(s) could not be fully inspected (for example, unsupported or malformed selector syntax).`);
  }

  if (findings.length === 0) {
    const message = "No issues were detected by the current checks. This is not proof that the document is fully valid, accessible, secure, or behaviorally correct.";
    findings.push({ severity: "info", title: "No issues detected by current checks", detail: message, code: "NO_FINDINGS" });
  }
  if (findings.length >= MAX_FINDINGS) {
    const message = `Finding limit (${MAX_FINDINGS}) reached; remaining checks may have additional results.`;
    findings[MAX_FINDINGS - 1] = { severity: "info", title: "Finding limit reached", detail: message, code: "LIMIT_REACHED" };
  }

  return {
    lines, elements: all.length, bytes, findings, scripts, styles, comments,
    parserNote: "DOMParser may repair malformed markup in its in-memory representation. Findings are advisory; the original source text is not reserialized or changed."
  };
}

function inspectEmbeddedCss(doc) {
  const messages = [];
  let externalStylesheets = 0;
  let unsupportedRules = 0;
  const records = new Map();
  let ruleOrder = 0;

  externalStylesheets = doc.querySelectorAll('link[rel~="stylesheet"][href]').length;

  const styleNodes = [...doc.querySelectorAll("style")];
  for (let styleIndex = 0; styleIndex < styleNodes.length; styleIndex++) {
    const styleNode = styleNodes[styleIndex];
    // Put the style into the parsed document to let the browser CSS parser interpret rules.
    if (!doc.head) continue;
    const temporaryStyle = doc.createElement("style");
    temporaryStyle.textContent = styleNode.textContent || "";
    doc.head.append(temporaryStyle);
    let rules;
    try {
      rules = [...(temporaryStyle.sheet?.cssRules || [])];
    } catch (error) {
      unsupportedRules++;
      temporaryStyle.remove();
      continue;
    }

    const walk = (ruleList, condition = "") => {
      for (const rule of ruleList) {
        ruleOrder++;
        if (rule.type === 1 && rule.selectorText && rule.style) {
          let matched = [];
          try {
            matched = [...doc.querySelectorAll(rule.selectorText)];
          } catch (error) {
            unsupportedRules++;
            continue;
          }
          for (const element of matched) {
            const elementName = describeElement(element);
            for (let i = 0; i < rule.style.length; i++) {
              const property = rule.style[i];
              const value = rule.style.getPropertyValue(property).trim();
              const important = rule.style.getPropertyPriority(property) === "important";
              const key = `${elementName}\u0000${property}`;
              const list = records.get(key) || [];
              list.push({ selector: rule.selectorText, property, value, important, condition, order: ruleOrder });
              records.set(key, list);
            }
          }
        } else if (rule.cssRules) {
          const label = rule.conditionText || rule.name || rule.constructor?.name || "nested rule";
          walk([...rule.cssRules], condition ? `${condition}; ${label}` : label);
        }
      }
    };

    walk(rules);
    temporaryStyle.remove();
  }

  for (const [key, list] of records) {
    const uniqueValues = new Set(list.map(item => `${item.value}\u0000${item.important ? "important" : "normal"}`));
    if (uniqueValues.size < 2) continue;
    const [element, property] = key.split("\u0000");
    const declarations = list.slice(0, 5).map(item =>
      `${item.selector} { ${item.property}: ${item.value}${item.important ? " !important" : ""} }${item.condition ? ` [${item.condition}]` : ""}`
    );
    messages.push(`Possible embedded-CSS conflict for ${element}, property “${property}”: ${declarations.join("; ")}${list.length > 5 ? "; …" : ""}. These selectors match the parsed element, but the analyzer does not determine the final computed value; confirm in browser DevTools → Computed.`);
    if (messages.length >= MAX_CSS_CONFLICTS) break;
  }

  return { messages, externalStylesheets, unsupportedRules };
}

function describeElement(element) {
  let result = element.tagName.toLowerCase();
  if (element.id) result += `#${element.id}`;
  const classes = [...element.classList].slice(0, 4);
  if (classes.length) result += classes.map(name => `.${name}`).join("");
  return result;
}


/**
 * Analyze supported source types without executing uploaded code.
 * HTML keeps the original DOM-based checks; other types get conservative
 * language-specific checks. Pattern matches are review prompts, not proof.
 */
function analyzeSource(source, fileName) {
  const ext = (String(fileName).match(/\.([^.]+)$/)?.[1] || "txt").toLowerCase();
  if (["html", "htm"].includes(ext)) return analyzeHtml(source, fileName);
  const lines = source ? source.split(/\r\n|\r|\n/).length : 0;
  const bytes = new TextEncoder().encode(source).length;
  let elements = 0;
  const findings = [];
  const seen = new Set();
  const add = (severity, title, detail, code = "SOURCE_AUDIT") => {
    const key = `${severity}\u0000${title}\u0000${detail}`;
    if (seen.has(key) || findings.length >= MAX_FINDINGS) return;
    seen.add(key); findings.push({severity, title, detail, code});
  };
  const scripts = {total:0, inline:0, external:0};
  const styles = {inline:0, external:0};
  let comments = 0;
  const lower = ext;
  if (["json"].includes(lower)) {
    try {
      const parsed = JSON.parse(source);
      elements = Array.isArray(parsed) ? parsed.length : parsed && typeof parsed === "object" ? Object.keys(parsed).length : 1;
      const kind = parsed === null ? "null" : Array.isArray(parsed) ? `array (${parsed.length} items)` : typeof parsed === "object" ? `object (${Object.keys(parsed).length} top-level keys)` : typeof parsed;
      add("info", "JSON syntax valid", `JSON.parse accepted this file. Top-level value: ${kind}.`, "JSON_VALID");
    } catch (error) {
      add("error", "Invalid JSON syntax", error && error.message ? error.message : String(error), "JSON_INVALID");
    }
  } else if (["js", "mjs", "cjs", "ts", "tsx"].includes(lower)) {
    if (/\beval\s*\(/.test(source)) add("warning", "Dynamic eval() call", "Review whether untrusted input can reach eval(); this is a pattern check, not a data-flow proof.");
    if (/\bnew\s+Function\s*\(/.test(source)) add("warning", "Dynamic Function constructor", "Review dynamic code generation and the trustworthiness of inputs.");
    if (/\binnerHTML\s*=/.test(source)) add("info", "Review HTML injection sink", "An innerHTML assignment exists. It is risky only if untrusted data can reach it; prefer textContent or DOM node creation when rendering plain text.", "INNERHTML_REVIEW");
    if (/catch\s*(?:\([^)]*\))?\s*\{\s*(?:\/\*[\s\S]*?\*\/\s*)?(?:\/\/[^\n]*\s*)?\}/.test(source)) add("warning", "Possibly empty catch block", "A catch block may swallow errors. Check whether a comment-only or empty handler needs explicit fallback, logging, or rethrow behavior.", "EMPTY_CATCH_REVIEW");
    if (/\bdebugger\s*;?/.test(source)) add("warning", "debugger statement", "Remove or justify debugger statements before release.");
    if (/\bconsole\.(?:log|debug|info)\s*\(/.test(source)) add("info", "Console output", "Review whether diagnostic output should remain in production.");
    if (/\bTODO\b|\bFIXME\b|\bHACK\b/.test(source)) add("info", "TODO/FIXME marker", "Review outstanding work markers.");
    if (/(?:api[_-]?key|secret|password|token)\s*[:=]\s*["'][^"']{8,}["']/i.test(source)) add("warning", "Possible hard-coded secret", "Pattern match only; verify whether the value is a real credential. Do not upload secrets to public repositories.");
    if (/\bfunction\b|=>|\bconst\b|\blet\b|\bclass\b/.test(source)) add("info", "Code patterns found", "Basic static patterns were inspected. This is not a complete JavaScript/TypeScript parser or type checker.");
  } else if (lower === "css") {
    const opens = (source.match(/\{/g) || []).length, closes = (source.match(/\}/g) || []).length;
    if (opens !== closes) add("error", "Unbalanced CSS braces", `Found ${opens} opening brace(s) and ${closes} closing brace(s). Strings and comments can affect this simple count; verify manually.`, "CSS_BRACE_BALANCE");
    if (/url\(\s*["']?\s*javascript\s*:/i.test(source)) add("warning", "Suspicious javascript: CSS URL", "Review this URL; browser behavior and support vary, and it should not be treated as a safe resource reference.");
    if (/!important\b/i.test(source)) add("info", "!important declarations found", "Review cascade overrides in context; !important is not automatically a defect.");
    if (/\/\*\s*(?:TODO|FIXME|HACK)\b/i.test(source)) add("info", "CSS TODO/FIXME marker", "Review outstanding style work markers.");
    try {
      const sheet = new CSSStyleSheet(); sheet.replaceSync(source);
      add("info", "CSS parser accepted source", "The browser CSS parser accepted the stylesheet. This does not guarantee intended visual behavior.", "CSS_PARSE_OK");
    } catch (error) {
      add("warning", "CSS parser could not fully parse source", error && error.message ? error.message : String(error), "CSS_PARSE_REVIEW");
    }
  } else if (lower === "py") {
    if (/\beval\s*\(/.test(source)) add("warning", "eval() usage", "Review dynamic evaluation and whether input can be controlled by an untrusted party.");
    if (/\bexec\s*\(/.test(source)) add("warning", "exec() usage", "Review dynamic execution and trust boundaries.");
    if (/\bpickle\.loads?\s*\(/.test(source)) add("warning", "Pickle deserialization", "Never deserialize pickle data from an untrusted source.");
    if (/\bexcept\s*(?:Exception|BaseException)?\s*(?:as\s+\w+)?\s*:\s*(?:pass\s*)?(?:\n|$)/.test(source)) add("warning", "Possible swallowed exception", "Review broad or empty exception handling; this pattern check can miss multiline cases.");
    if (/\bTODO\b|\bFIXME\b|\bHACK\b/.test(source)) add("info", "TODO/FIXME marker", "Review outstanding work markers.");
    if (/(?:api[_-]?key|secret|password|token)\s*=\s*["'][^"']{8,}["']/i.test(source)) add("warning", "Possible hard-coded secret", "Pattern match only; verify whether the value is a real credential.");
    add("info", "Python syntax not executed", "This browser-only tool does not run Python or claim full syntax validation. Use a Python interpreter/linter for authoritative syntax and type checks.", "PYTHON_SCOPE");
  } else {
    add("info", "Text-only checks", `No language-specific parser is configured for .${lower || "txt"}; only file integrity and conservative text cleanup are available.`);
  }
  if (findings.length === 0) add("info", "No configured pattern detected", "This is not proof of correctness, security, accessibility, or runtime behavior.", "NO_FINDINGS");
  return {lines, elements, bytes, findings, scripts, styles, comments, parserNote: `File type: .${lower || "txt"}. Uploaded code was not executed. Checks are advisory and parser coverage is limited.`, fileKind: lower};
}

function renderReport(report) {

  metrics.replaceChildren();

}

/** Format byte counts for the workbench UI. */
function formatBytes(bytes) {
  if (!Number.isFinite(bytes) || bytes < 0) return "—";
  if (bytes < 1024) return `${bytes} B`;
  const units = ["KiB", "MiB", "GiB"];
  let value = bytes / 1024;
  let index = 0;
  while (value >= 1024 && index < units.length - 1) { value /= 1024; index++; }
  return `${value.toFixed(value >= 10 ? 1 : 2)} ${units[index]}`;
}

  return {output,changes,changed:output!==original,bytesBefore:new TextEncoder().encode(original).length,bytesAfter:new TextEncoder().encode(output).length};
}

// CSS usage analysis: source-preserving scanner for simple top-level rules only.
// Deliberately excludes nested at-rules, dynamic pseudo-classes, and complex syntax from auto-removal.
function scanEmbeddedCssUsage(source) {
  const doc = new DOMParser().parseFromString(source, "text/html");
  const rules = [];
  const styleTagRe = /<style\b[^>]*>([\s\S]*?)<\/style\s*>/gi;
  let tagMatch;
  const dynamicPseudo = /:(?:hover|focus|focus-within|focus-visible|active|visited|checked|disabled|enabled|target|has|not|is|where|empty|first-child|last-child|nth-child|nth-of-type|before|after|root|valid|invalid|open|popover-open)\b/i;
  while ((tagMatch = styleTagRe.exec(source))) {
    const css = tagMatch[1];
    const cssStart = tagMatch.index + tagMatch[0].indexOf(tagMatch[1]);
    // A tiny brace-aware scanner: ignores comments and quoted braces; only top-level blocks are candidates.
    let i = 0;
    while (i < css.length) {
      while (i < css.length && /\s/.test(css[i])) i++;
      if (css.startsWith('/*', i)) { const end = css.indexOf('*/', i+2); i = end < 0 ? css.length : end+2; continue; }
      const open = css.indexOf('{', i); if (open < 0) break;
      const head = css.slice(i, open).trim();
      let depth = 1, j = open+1, quote = '', comment = false;
      for (; j < css.length && depth; j++) {
        const c = css[j], n = css[j+1];
        if (comment) { if (c==='*' && n==='/') { comment=false; j++; } continue; }
        if (quote) { if (c==='\\') { j++; continue; } if (c===quote) quote=''; continue; }
        if (c==='/' && n==='*') { comment=true; j++; continue; }
        if (c==='"' || c==="'") { quote=c; continue; }
        if (c==='{') depth++; else if (c==='}') depth--;
      }
      if (depth) break;
      const end = j;
      if (head && !head.startsWith('@') && !head.includes('{') && !head.includes('}')) {
        const selectors = head.split(',').map(x=>x.trim()).filter(Boolean);
        if (selectors.length && selectors.every(sel=>!dynamicPseudo.test(sel) && !/[+~]|\\\\|:/.test(sel))) {
          let matched = false, parseable = true;
          for (const selector of selectors) { try { if (doc.querySelector(selector)) matched = true; } catch { parseable = false; } }
          if (parseable && !matched) {
            const full = css.slice(i,end);
            rules.push({id:rules.length, selector:head, code:full, start:cssStart+i, end:cssStart+end, line:source.slice(0,cssStart+i).split(/\r\n|\r|\n/).length, related:selectors.length, reason:'No matching element in the parsed HTML; dynamic use cannot be ruled out.'});
          }
        }
      }
      // Do not descend into nested rules. Move past this block to avoid mistaking declarations for rules.
      i = end;
    }
  }
  // Group rules with shared selector tokens/classes/IDs to expose relationships rather than a flat count.
  const tokenMap = new Map();
  for (const r of rules) {
    const tokens = [...r.selector.matchAll(/[.#][A-Za-z_][\w-]*/g)].map(m=>m[0]);
    r.tokens = tokens;
    for (const t of tokens) tokenMap.set(t,[...(tokenMap.get(t)||[]),r.id]);
  }
  for (const r of rules) r.relatedIds = [...new Set(r.tokens.flatMap(t=>tokenMap.get(t)||[]).filter(id=>id!==r.id))];
  return {rules, groups:[...tokenMap.entries()].filter(([,ids])=>ids.length>1).map(([token,ids])=>({token,ids:[...new Set(ids)]}))};
}
